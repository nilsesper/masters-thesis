%%% command definitions
% configure chapter headings:
%  1: vspace from top of page in pt (default: 50)
%  3: vspace after chapter heading in pt (default: 40)
%  2: heading above number and chapter name (e.g. "Appendix"), if empty: no extra heading
\makeatletter
\newcommand{\configchapterheading}[3]{
    \renewcommand{\@makechapterhead}[1]{
      \ifthenelse{\equal{#1}{}}{ \vspace*{50\p@} }{ \vspace*{#1\p@} }
      {\parindent \z@ \raggedright \normalfont
        \interlinepenalty\@M
        \ifthenelse{\equal{#3}{}}{}{
            \Huge\bfseries  #3 \par\nobreak \vskip 20\p@
        }
        \Huge\bfseries  \thechapter \quad ##1 \par\nobreak
        \ifthenelse{\equal{#2}{}}{ \vskip 40\p@ }{ \vskip #2\p@ }
    }}
}
\makeatother

%%% PAGE STYLES (headers, footers) %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
% clear page: "empty"
% only page numbers at bottom
\renewpagestyle{plain} {
    \sethead[][][]{}{}{} % header
    \setfoot[\thepage][][]{}{}{\thepage} % footer
}
% chapter name in header without chapter number
\newpagestyle{chapterheadernonumber}{%
  \setheadrule{0.5pt} % line
  \sethead[\chaptertitle][][]{}{}{\chaptertitle} % header
  \setfoot[\thepage][][]{}{}{\thepage} % footer
}
% chapter name and number in header
\newpagestyle{chapterheader}{%
  \setheadrule{0.5pt} % line
  \sethead[\thechapter~$\;\;$~\chaptertitle][][]{}{}{\chaptertitle~$\;\;$~\thechapter} % header
  \setfoot[\thepage][][]{}{}{\thepage} % footer
}
% section and chapter name and number in header
\newpagestyle{sectionheader}{%
  \setheadrule{0.5pt} % line
  \sethead[\thesection~$\;\;$~\chaptertitle~$-$~\sectiontitle][][]{}{}{\chaptertitle~$-$~\sectiontitle~$\;\;$~\thesection} % header
  \setfoot[\thepage][][]{}{}{\thepage} % footer
}
% list of acronyms header
\newpagestyle{loaheader}{%
  \setheadrule{0.5pt} % line
  \sethead[List of Acronyms][][]{}{}{List of Acronyms} % header
  \setfoot[\thepage][][]{}{}{\thepage} % footer
}
% acknowledgements header
\newpagestyle{ackheader}{%
  \setheadrule{0.5pt} % line
  \sethead[Acknowledgements][][]{}{}{Acknowledgements} % header
  \setfoot[\thepage][][]{}{}{\thepage} % footer
}
% abstract header
\newpagestyle{abstractheader}{%
  \setheadrule{0.5pt} % line
  \sethead[Abstract][][]{}{}{Abstract} % header
  \setfoot[\thepage][][]{}{}{\thepage} % footer
}
% bibliography header
\newpagestyle{bibheader}{%
  \setheadrule{0.5pt} % line
  \sethead[Bibliography][][]{}{}{Bibliography} % header
  \setfoot[\thepage][][]{}{}{\thepage} % footer
}
